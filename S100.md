The algorithm used for the S100 certificate is ECDSA with SHA384.
We wil use the keystore already created "secom_{{env}}_keystore.jks" to store the certificates and private keys.

You must first create s100_labnum_certificates directory to store the certificates and private keys if it does not exist.
```
cd certificates
mkdir s100_labnum_certificates
cd s100_labnum_certificates
```


# create S100 ROOT trusted certificate
```
openssl ecparam -out ecparam.txt -name secp384r1 
openssl req -x509 -sha384 -nodes -days 365 -newkey ec:ecparam.txt -keyout s100_root.key -out s100_root.crt
```

Use following:
```
Country Name (2 letter code) [AU]:XX
State or Province Name (full name) [Some-State]:XXXXX
Locality Name (eg, city) []:
Organization Name (eg, company) [Internet Widgits Pty Ltd]:XXXXX
Organizational Unit Name (eg, section) []:
Common Name (e.g. server FQDN or YOUR name) []:PING
Email Address []:
```

# import trusted S100 ROOT certificate as CA certificate in keystore
```
keytool -import -trustcacerts -alias s100_root -keystore ../secom_{{env}}_keystore.jks -file s100_root.crt -storepass SECOM_{{env}}_KEYSTORE_PASSWORD
```

# create CSR for SHOM
```
openssl ecparam -out shom-param.txt -name secp384r1 
openssl req -out shom.csr -new -newkey ec:shom-param.txt -nodes -keyout shom_private.key -sha384
```


Use following:
```
Country Name (2 letter code) [AU]:XX
State or Province Name (full name) [Some-State]:XXXXXX
Locality Name (eg, city) []:
Organization Name (eg, company) [Internet Widgits Pty Ltd]:XXXX
Organizational Unit Name (eg, section) []:
Common Name (e.g. server FQDN or YOUR name) []:XXXX
Email Address []:

Please enter the following 'extra' attributes
to be sent with your certificate request
A challenge password []:xxxxxxx
An optional company name []:
```

# split public key from private key
```
openssl ec -outform pem -in shom_private.key -out shom_public.txt -pubout
```

# create signed SHOM certificate by CA from CSR
```
openssl x509 -req -in shom.csr -sha384 -CA s100_root.crt -CAkey s100_root.key -CAcreateserial -out shom.crt
```

# check SHOM certificate against CA (PING ROOT)
```
openssl verify -verbose -CAfile s100_root.crt shom.crt
```

# import SHOM certificate in keystore
```
keytool -import -alias shom -file shom.crt -keystore ../secom_{{env}}_keystore.jks -storepass SECOM_{{env}}_KEYSTORE_PASSWORD
```

# import private key to keystore
```
openssl pkcs12 -export -in s100_root.crt -inkey s100_root.key -out s100_root.p12 -name s100_root -CAfile s100_root.crt -caname s100_root
keytool -importkeystore -srckeystore s100_root.p12 -srcstoretype PKCS12 -destkeystore ../secom_{{env}}_keystore.jks -deststoretype JKS -srcstorepass SECOM_{{env}}_KEYSTORE_PASSWORD -deststorepass SECOM_{{env}}_KEYSTORE_PASSWORD
```

```
openssl pkcs12 -export -in shom.crt -inkey shom_private.key -out shom.p12 -name shom -CAfile s100_root.crt -caname s100_root
keytool -importkeystore -srckeystore shom.p12 -srcstoretype PKCS12 -destkeystore ../secom_{{env}}_keystore.jks -deststoretype JKS -srcstorepass SECOM_{{env}}_KEYSTORE_PASSWORD -deststorepass SECOM_{{env}}_KEYSTORE_PASSWORD
```

# list certificates
```
keytool -list -rfc -keystore ../secom_{{env}}_keystore.jks -storepass SECOM_{{env}}_KEYSTORE_PASSWORD
```
